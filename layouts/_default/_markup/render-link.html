{{- $destination := .Destination -}} {{- $isHttp := or (hasPrefix $destination "http://") (hasPrefix $destination "https://") -}} {{- $baseHost := (urls.Parse site.BaseURL).Host -}} {{- $destHost := "" -}} {{- if $isHttp -}} {{- $destHost = (urls.Parse $destination).Host -}} {{- end -}} {{- $isExternal := and $isHttp (ne $destHost "") (ne $destHost $baseHost) -}} {{- $linkClass := cond $isExternal "external-link" "internal-link" -}} {{ .Text }}{{- if $isExternal -}}{{ partial "icons/external-link.html" . }}{{- end -}}
